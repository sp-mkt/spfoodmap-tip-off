/** * 🍽️ 스슐랭 맛집 제보 — Apps Script 백엔드 (정보 수급 및 면책 동의 반영 버전) * * 검색·좌표는 제보 페이지에서 카카오 지도 SDK가 브라우저에서 직접 처리합니다. * 따라서 이 스크립트는 "제보 데이터를 구글 시트에 저장"하는 역할만 합니다. */ const SHEET_NAME = "제보"; // 기존 지도 페이지 컬럼명 유지 + 제보자 상세 정보(회사명, 근무지점, 성함, 연락처) 추가 const HEADERS = [ "제보일시", "노출", "이름", "이모지", "카테고리", "한줄평", "가격대", "태그", "위도", "경도", "네이버맵URL", "주소", "제보자(닉네임)", "회사명", "근무지점", "성함", "연락처" ]; function doPost(e) { try { const d = JSON.parse(e.postData.contents); if (!d.name || !d.category || !d.review) { return jsonOut({ ok: false, error: "필수 항목 누락" }); } const sheet = getSheet(); sheet.appendRow([ Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss"), false, // 노출: 검토 전 기본 FALSE String(d.name).slice(0, 100), "", // 이모지: 검토 시 입력 String(d.category).slice(0, 30), String(d.review).slice(0, 100), String(d.price || "").slice(0, 20), String(d.tags || "").slice(0, 100), d.lat || "", d.lng || "", String(d.mapUrl || "").slice(0, 300), String(d.address || "").slice(0, 200), String(d.nickname || "").slice(0, 50), // 성함 데이터가 여기에도 매핑되어 백업됩니다. String(d.company || "").slice(0, 50), // [새로 추가] 회사명 String(d.branch || "").slice(0, 50), // [새로 추가] 근무 지점 String(d.nickname || "").slice(0, 50), // [새로 추가] 제보자 성함 String(d.contact || "").slice(0, 30) // [새로 추가] 연락처 ]); return jsonOut({ ok: true }); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } } // 배포 점검용 (브라우저로 웹앱 URL 직접 열었을 때 동작 확인) function doGet() { return jsonOut({ ok: true, msg: "스슐랭 맛집 제보 수집기가 정상 동작 중입니다." }); } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) sheet = ss.insertSheet(SHEET_NAME); // 만약 시트가 완전히 비어있다면 헤더 행을 새로 생성합니다. if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight("bold").setBackground("#f1f3f5"); } else { // [중요] 기존 시트에 열이 이미 추가되어 있다면, 새 헤더명으로 첫 행만 갱신해 줍니다. sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); } return sheet; } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** * 스프레드시트가 열릴 때 상단에 커스텀 메뉴를 추가합니다. */ function onOpen() { const ui = SpreadsheetApp.getUi(); ui.createMenu('⚙️ 자동화 매크로') .addItem('카카오맵 URL 생성하기', 'makeKakaoMapUrl') .addToUi(); }
